# ============================================================
# EXPERIMENT 8 — OPEN vSWITCH AND LINUX BRIDGE
# Ubuntu 18.04
# ============================================================

# 1. Check system and network information
uname -a
ifconfig
ip addr
ip route

sudo apt-get update

sudo apt-get install -y bridge-utils
sudo modprobe bridge
sudo modprobe br_netfilter

lsmod | grep bridge

sudo apt-get install -y openvswitch-switch

sudo systemctl status openvswitch-switch

sudo systemctl start openvswitch-switch

sudo systemctl enable openvswitch-switch

# ============================================================
# PART A — LINUX BRIDGE
# ============================================================
sudo brctl addbr br-cloud

sudo ip link set br-cloud up

sudo ip link add veth1 type veth peer name veth2

sudo ip link add veth3 type veth peer name veth4

sudo ip netns add VRF1
sudo ip netns add VRF2

sudo ip link set veth1 netns VRF1
sudo ip link set veth3 netns VRF2

sudo ip netns exec VRF1 ip link set lo up
sudo ip netns exec VRF2 ip link set lo up

sudo ip netns exec VRF1 ip addr add 10.10.10.1/24 dev veth1
sudo ip netns exec VRF2 ip addr add 10.10.10.2/24 dev veth3

sudo ip netns exec VRF1 ip link set veth1 up
sudo ip netns exec VRF2 ip link set veth3 up

sudo ip link set veth2 up
sudo ip link set veth4 up

sudo brctl addif br-cloud veth2
sudo brctl addif br-cloud veth4

brctl show

ip netns list

sudo ip netns exec VRF1 ip addr

sudo ip netns exec VRF2 ip addr

sudo ip netns exec VRF1 ping -c 4 10.10.10.2

# ============================================================
# PART B — OPEN vSWITCH
# ============================================================

sudo ovs-vsctl show

sudo ovs-vsctl add-br vSwitch1

sudo ip link add ovs-veth1 type veth peer name ovs-veth2

sudo ip link set ovs-veth1 up
sudo ip link set ovs-veth2 up

sudo ovs-vsctl add-port vSwitch1 ovs-veth1

sudo ip link set vSwitch1 up

sudo ovs-vsctl show

sudo ovs-vsctl list-br

sudo ovs-vsctl list-ports vSwitch1
